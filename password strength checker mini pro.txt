import React, { useState } from "react";

function PasswordSecurityAnalyzer() {
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);

  const commonPasswords = [
    "123456",
    "password",
    "qwerty",
    "admin",
    "welcome",
    "12345678",
  ];

  const analyzePassword = (pwd) => {
    let score = 0;
    let issues = [];

    if (pwd.length >= 8) score++;
    else issues.push("Password should be at least 8 characters");

    if (/[A-Z]/.test(pwd)) score++;
    else issues.push("Add an uppercase letter");

    if (/[a-z]/.test(pwd)) score++;
    else issues.push("Add a lowercase letter");

    if (/[0-9]/.test(pwd)) score++;
    else issues.push("Add a number");

    if (/[^A-Za-z0-9]/.test(pwd)) score++;
    else issues.push("Add a special character");

    if (commonPasswords.includes(pwd.toLowerCase())) {
      issues.push("Common password detected");
      score = Math.max(score - 2, 0);
    }

    let strength = "Weak";
    let color = "red";

    if (score >= 3 && score < 5) {
      strength = "Medium";
      color = "orange";
    }

    if (score === 5) {
      strength = "Strong";
      color = "green";
    }

    const securityScore = Math.min(score * 20, 100);

    let crackTime = "Few Seconds";

    if (score === 2) crackTime = "Few Minutes";
    if (score === 3) crackTime = "Several Days";
    if (score === 4) crackTime = "Several Years";
    if (score === 5) crackTime = "100+ Years";

    return {
      strength,
      color,
      issues,
      securityScore,
      crackTime,
    };
  };

  const generateSuggestion = () => {
    return "Acharya0399#Secure";
  };

  const result = analyzePassword(password);

  return (
    <div
      style={{
        maxWidth: "500px",
        margin: "50px auto",
        padding: "20px",
        border: "1px solid #ddd",
        borderRadius: "10px",
        textAlign: "center",
        fontFamily: "Arial",
      }}
    >
      <h2>🔐 Smart Password Security Analyzer</h2>

      <input
        type={showPassword ? "text" : "password"}
        placeholder="Enter Password"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
        style={{
          width: "90%",
          padding: "10px",
          marginBottom: "10px",
        }}
      />

      <br />

      <button
        onClick={() => setShowPassword(!showPassword)}
        style={{
          padding: "8px 15px",
          cursor: "pointer",
        }}
      >
        {showPassword ? "Hide Password" : "Show Password"}
      </button>

      {password && (
        <div style={{ marginTop: "20px" }}>
          <h3 style={{ color: result.color }}>
            Strength: {result.strength}
          </h3>

          <h4>Security Score: {result.securityScore}/100</h4>

          <h4>⏳ Estimated Crack Time</h4>
          <p>{result.crackTime}</p>

          <h4>⚠ Issues Found</h4>
          <ul style={{ textAlign: "left" }}>
            {result.issues.length > 0 ? (
              result.issues.map((issue, index) => (
                <li key={index}>{issue}</li>
              ))
            ) : (
              <li>No issues detected</li>
            )}
          </ul>

          {result.strength !== "Strong" && (
            <>
              <h4>💡 Suggested Password</h4>
              <p>{generateSuggestion()}</p>
            </>
          )}
        </div>
      )}
    </div>
  );
}

export default PasswordSecurityAnalyzer;
